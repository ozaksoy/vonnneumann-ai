from vonnneumann import AgentRuntime, LLMCluster

runtime = AgentRuntime(cluster=LLMCluster.HYBRID_ENTERPRISE)

@runtime.agent(name="DataAnalyst", isolation="strict")
async def process_context(ctx):
    response = await ctx.orchestrate(
        model="claude-3-5-sonnet",
        context_window=200000,
        strict_latency_ms=85
    )
    return response.stream()
import { VonnNeumannRuntime } from '@vonnneumann/sdk';

const runtime = new VonnNeumannRuntime({ region: 'global-edge' });

export async function executeAgent(payload) {
  const agent = runtime.createAgent({ model: 'claude-3-5-sonnet' });
  return await agent.run({ input: payload, maxLatencyMs: 85 });
}

Frequently Asked Questions

Technical questions regarding our runtime and deployment model.

How does Vonn Neumann achieve sub-100ms orchestration?

By compiling agent prompt trees into compiled WASM state machines and utilizing edge-cached connection pools directly to Claude 3.5 Sonnet endpoints.

Is user data stored or used for training?

No. Vonn Neumann operates under strict Zero Data Retention (ZDR) guarantees. All agent state contexts are wiped immediately post-execution.

Which LLM providers are supported?

While primary optimizations are tailored for Anthropic's Claude 3.5 Sonnet and Haiku models, hybrid cluster failovers can be configured via our SDK.
